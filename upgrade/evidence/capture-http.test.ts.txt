// Evidence capture (upgrade run, not a product test). Copy to test/e2e/zz-capture-http.test.ts (drop .txt), run
// `CAPTURE_LABEL=before pnpm vitest run --project e2e zz-capture-http`, then delete the copy.
import { mkdirSync, writeFileSync } from 'node:fs'
import { describe, expect, inject, it } from 'vitest'

const host = inject('e2eHost')
const label = process.env.CAPTURE_LABEL ?? 'before'
const SKIP = new Set(['date', 'content-length', 'etag', 'last-modified'])
const DYNAMIC = /^(x-request-id|request-id|x-nuxt-.*|cf-ray)$/i

const cases = [
  ['a', 'unknown page under a cafe', '/c/nuk/definitely-not-a-page'],
  ['b', 'unknown cafe API route', '/api/c/nuk/definitely-not-a-route'],
  ['c', 'unauthenticated admin API GET', '/api/c/nuk/admin/orders'],
  ['d', 'SSR redirect (server/middleware/30.old-addresses) with percent-encoded query', '/admin/orders?q=caf%C3%A9%20au%20lait&tag=a%26b&next=%2Ftable%2Fx%3Fy%3D1'],
  ['d2', 'admin page without a session (auth is a client-side redirect; no Location expected)', '/c/nuk/admin/orders?q=caf%C3%A9%20au%20lait&tag=a%26b&next=%2Ftable%2Fx%3Fy%3D1'],
  ['e', 'unknown table token', '/table/not-a-real-token'],
] as const

function shape(value: unknown, depth = 0): unknown {
  if (Array.isArray(value)) return value.length ? [shape(value[0], depth + 1)] : []
  if (value && typeof value === 'object') {
    return depth > 2 ? '{…}' : Object.fromEntries(Object.entries(value).map(([k, v]) => [k, shape(v, depth + 1)]))
  }
  return typeof value
}

describe('capture', () => {
  it('records the five cases', async () => {
    let out = `# HTTP evidence — ${label}\n\nServer: e2e build (\`.output/server/index.mjs\`), seed cafe \`nuk\`. Dynamic header values are masked.\n`
    for (const [id, title, path] of cases) {
      const res = await fetch(host + path, { redirect: 'manual', headers: { accept: path.startsWith('/api') ? 'application/json' : 'text/html' } })
      const text = await res.text()
      const headers = [...res.headers.entries()].filter(([k]) => !SKIP.has(k)).sort(([a], [b]) => a.localeCompare(b))
        .map(([k, v]) => `${k}: ${DYNAMIC.test(k) ? '<dynamic>' : v}`)
      let body: string
      try { body = `JSON shape: ${JSON.stringify(shape(JSON.parse(text)))}` }
      catch { body = `non-JSON (${res.headers.get('content-type')}): title=${/<title>([^<]*)<\/title>/.exec(text)?.[1] ?? '(none)'}, has __NUXT__ payload=${/__NUXT_DATA__|window\.__NUXT__/.test(text)}` }
      out += `\n## (${id}) ${title}\n\n\`GET ${path}\` → **${res.status}**${res.headers.get('location') ? ` Location: \`${res.headers.get('location')}\`` : ''}\n\n\`\`\`\n${headers.join('\n')}\n\`\`\`\n\nBody: ${body}\n`
    }
    mkdirSync(`upgrade/evidence/${label}`, { recursive: true })
    writeFileSync(`upgrade/evidence/${label}/http.md`, out)
    expect(out).toContain('(e)')
  })
})
